function clock() {
  let at = 100, next = 1;
  const timers = new Map(), scheduled = [], cleared = [];
  return {
    now: () => at,
    setTimer(fn, ms) { const id = next++; timers.set(id, {fn, due: at + ms}); scheduled.push(ms); return id; },
    clearTimer(id) { cleared.push(id); timers.delete(id); },
    advance(ms) {
      at += ms;
      for (const [id, timer] of [...timers]) if (timer.due <= at) { timers.delete(id); timer.fn(); }
    },
    timers, scheduled, cleared,
  };
}
const costAddress = extra => address('qa&qa-memory=deferred-bounded-test&qa-player-deferred-read=1&qa-owned-cleanup=1' + (extra ?? ''));
const originalAddress = () => address('qa&qa-memory=deferred-bounded-original&qa-original-readback=1&qa-owned-cleanup=1');
const trace = e => e.draws.map(({state, motion, input, command, time}) => ({state, motion, input, command, time}));
function setup(e, {original = false, bounded = true, drain = 'normal'} = {}) {
  const time = clock();
  let factories = 0, waits = 0;
  e.testDeps.ownedCleanupTestOptions = {now: time.now, setTimer: time.setTimer, clearTimer: time.clearTimer, timeoutMs: 5};
  e.testDeps.createOwnedCleanupTracker = options => {
    factories++;
    const tracker = createOwnedCleanupTracker(options);
    return {
      follow: tracker.follow,
      snapshot: tracker.snapshot,
      wait() {
        waits++;
        const waiting = tracker.wait();
        if (drain === 'resolve') e.pending.resolve();
        if (drain === 'timeout') time.advance(5);
        return waiting;
      },
    };
  };
  e.testDeps.probeOriginalReadback = original
    ? () => ({complete: true, stable: true, injected: true})
    : () => { throw Error('cost mode must not invoke original readback probes'); };
  const options = {
    ...gate(original ? originalAddress() : costAddress()),
    ...(original ? {originalReadbackProbe: true} : {playerDeferredRead: true}),
    boundedOwnedCleanup: bounded,
    document: e.document,
    testDeps: e.testDeps,
  };
  return {options, time, factories: () => factories, waits: () => waits};
}
function assertFrozenTree(value) {
  if (value === null || typeof value !== 'object') return;
  assert.equal(Object.isFrozen(value), true);
  assert.equal(ArrayBuffer.isView(value), false);
  for (const child of Object.values(value)) assertFrozenTree(child);
}

test('old bounded original and disabled-bounded deferred failure paths retain full frozen-before reports, clocks and call order', async () => {
  for (const original of [true, false]) {
    const a = environment('render-failure'), b = environment('render-failure');
    if (!original) { deferredInstaller(a); deferredInstaller(b); }
    const x = setup(a, {original, bounded: original}), y = setup(b, {original, bounded: original});
    const old = await failure(before.compareCanvasStateCombat(a.mainRenderer, x.options));
    const current = await failure(compareCanvasStateCombat(b.mainRenderer, y.options));
    assert.equal(current.message, old.message);
    assert.deepEqual(current.report, old.report);
    assert.deepEqual(current.cleanup, old.cleanup);
    assert.deepEqual(b.events, a.events);
    assert.deepEqual(trace(b), trace(a));
    assert.equal(x.factories(), original ? 1 : 0);
    assert.equal(y.factories(), original ? 1 : 0);
    assert.equal(Object.hasOwn(current.report, 'deferredBoundedCost'), false);
    mainUnchanged(a); mainUnchanged(b);
    ownedImagesAndAssetsRetired(a); ownedImagesAndAssetsRetired(b);
  }
});

test('bounded deferred cost requires exclusive exact flags and genuine memory before any owner, DOM or factory access', async () => {
  const invalid = [
    [address('qa&qa-memory=d&qa-player-deferred-read=1'), {}],
    [address('qa&qa-memory=d&qa-player-deferred-read=1&qa-owned-cleanup=0'), {}],
    [costAddress('&qa-player-deferred-read-verify=1'), {}],
    [costAddress(), {playerDeferredReadVerify: true}],
    [costAddress(), {originalReadbackProbe: true}],
    [costAddress(), {originalAttribution: true}],
    [costAddress(), {playerAttribution: true}],
    [costAddress(), {playerPreparation: true}],
    [costAddress(), {playerAlphaRows: true}],
    [new URL('http://localhost:8806/?qa&qa-memory=d&qa-player-deferred-read=1&qa-owned-cleanup=1'), {}],
    [new URL('https://localhost:8851/?qa&qa-memory=d&qa-player-deferred-read=1&qa-owned-cleanup=1'), {}],
  ];
  for (const [location, extra] of invalid) {
    let reads = 0;
    const options = {...gate(location), playerDeferredRead: true, boundedOwnedCleanup: true, ...extra};
    for (const key of ['document', 'testDeps']) Object.defineProperty(options, key, {get() { reads++; throw Error('gate must precede dependencies'); }});
    const renderer = new Proxy({}, {
      get() { reads++; throw Error('gate must precede owner reads'); },
      getOwnPropertyDescriptor() { reads++; throw Error('gate must precede owner descriptors'); },
    });
    const result = await compareCanvasStateCombat(renderer, options);
    assert.equal(result.report.available, false);
    assert.equal(result.report.complete, false);
    assert.equal(reads, 0);
  }
  let reads = 0;
  const location = costAddress();
  const result = await compareCanvasStateCombat(new Proxy({}, {get() { reads++; throw Error('fake memory owner'); }}), {
    location, query: location.searchParams, storage: new Map(),
    playerDeferredRead: true, boundedOwnedCleanup: true,
    get testDeps() { reads++; throw Error('fake memory factory'); },
  });
  assert.equal(result.report.available, false);
  assert.equal(reads, 0);
});

test('timeout preserves frozen primary failure and late retirement, while stop/read faults remain errors instead of successful cleanup', async () => {
  const e = environment('late-model-failure'), handles = deferredInstaller(e), configured = setup(e, {drain: 'timeout'});
  const error = await failure(compareCanvasStateCombat(e.mainRenderer, configured.options));
  assert.equal(error.message, 'controlled canvas combat render failure');
  assert.equal(error.cleanup.complete, false);
  assert.equal(error.cleanup.drain.timedOut, true);
  assert.equal(error.cleanup.drain.active, 1);
  assert.equal(error.cleanup.drain.pending[0].label.key, 'enemy-elite');
  assert.equal(handles[0].stops, 1);
  assertFrozenTree(error.cleanup);
  const saved = JSON.stringify(error.report), rig = e.renderers[0].liveEnemies, beforeState = state(e.advances[0].game);
  e.pending.resolve();
  await turn(); await turn();
  assert.equal(rig.lateModelsDisposed, 1);
  assert.equal(rig.pending.size, 0);
  assert.equal(rig.models.size, 0);
  assert.equal(JSON.stringify(error.report), saved);
  assert.deepEqual(state(e.advances[0].game), beforeState);
  mainUnchanged(e); ownedImagesAndAssetsRetired(e);

  const stopped = environment('render-failure');
  const stoppedHandles = deferredInstaller(stopped, {mode: 'stop-failure'});
  const failedStop = await failure(compareCanvasStateCombat(stopped.mainRenderer, setup(stopped).options));
  assert.equal(failedStop.message, 'controlled canvas combat render failure');
  assert.equal(failedStop.cleanup.complete, false);
  assert.equal(failedStop.cleanup.drain.success, true);
  assert.ok(failedStop.cleanup.errors.some(row => row.where === 'playerDeferredRead.stop'));
  assert.ok(stoppedHandles.every(handle => handle.stops === 1));
  assertFrozenTree(failedStop.cleanup);
  mainUnchanged(stopped); ownedImagesAndAssetsRetired(stopped);

  const read = environment(), OriginalRenderer = read.testDeps.Renderer;
  read.testDeps.Renderer = class extends OriginalRenderer {
    constructor(...args) {
      super(...args);
      this.ctx.getImageData = () => { throw Error('controlled final R0 read failure'); };
    }
  };
  const readHandles = deferredInstaller(read);
  const failedRead = await failure(compareCanvasStateCombat(read.mainRenderer, setup(read).options));
  assert.equal(failedRead.message, 'controlled final R0 read failure');
  assert.equal(read.draws.length, 40);
  assert.equal(failedRead.cleanup.drain.success, true);
  assert.equal(failedRead.cleanup.complete, true);
  assert.ok(readHandles.every(handle => handle.stops === 1));
  assertFrozenTree(failedRead.cleanup);
  mainUnchanged(read); ownedImagesAndAssetsRetired(read);
});

test('aborting preloading still drains owned late work without drawing, and reports stay fixed after cleanup', async () => {
  const e = environment('deferred'), controller = new AbortController(), configured = setup(e, {drain: 'resolve'});
  e.setController(controller);
  deferredInstaller(e);
  const running = compareCanvasStateCombat(e.mainRenderer, {...configured.options, signal: controller.signal});
  running.catch(() => {});
  await turn();
  controller.abort();
  const error = await failure(running);
  assert.match(error.message, /aborted/);
  assert.equal(e.draws.length, 0);
  assert.equal(e.advances.length, 0);
  assert.equal(error.cleanup.complete, true);
  assert.equal(error.cleanup.drain.success, true);
  assert.equal(error.cleanup.drain.active, 0);
  assert.ok(error.cleanup.drain.scheduled > 0);
  assertFrozenTree(error.cleanup);
  const saved = JSON.stringify(error.report);
  await turn(); await turn();
  assert.equal(JSON.stringify(error.report), saved);
  mainUnchanged(e); ownedImagesAndAssetsRetired(e);
});
