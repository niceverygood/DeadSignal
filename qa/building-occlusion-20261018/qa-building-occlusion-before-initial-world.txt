export function legalPoint(game,wall,r,face){
 for(const extra of [.5,1,2,3,4,5,10,20,35,50,80])for(const slide of face==='S'&&r<=14?[-.44,0,-.2,.2,-.35,.35]:face.length===1?[0,-.2,.2,-.35,.35]:[0]){
  const gap=r/Math.SQRT2+extra,cx=wall.x+wall.w*(.5+slide),cy=wall.y+wall.h*(.5+slide),x=face.includes('W')?wall.x-(face.length===1?r+extra:gap):face.includes('E')?wall.x+wall.w+(face.length===1?r+extra:gap):cx,y=face.includes('N')?wall.y-(face.length===1?r+extra:gap):face.includes('S')?wall.y+wall.h+(face.length===1?r+extra:gap):cy;
  if(!game.blocked(x,y,r))return{x,y,r,blocked:false,buildingOrder:buildingActorOrder(wall,{x,y,r})};
 }
 throw Error('building-occlusion-no-legal-'+face+'-position');
}
function rearPoint(game,wall,r,target){
 const choices=[];for(const t of [.8,.65,.5,.35,.2])for(const gap of [r+5,r+15,r+30]){choices.push({x:wall.x+wall.w*t,y:wall.y-gap});choices.push({x:wall.x-gap,y:wall.y+wall.h*t});}
 const legal=choices.filter(p=>!game.blocked(p.x,p.y,r)&&buildingActorOrder(wall,{...p,r})===-1&&Math.hypot(p.x-target.x,p.y-target.y)>target.r+r+8);

}
export function playerAnchor(game,wall,target,partner,{nearTarget=false}={}){

 // Prefer a separate screen-X column, then retain the same actual collision
 // and radius checks. The production sorter independently certifies bounds.
