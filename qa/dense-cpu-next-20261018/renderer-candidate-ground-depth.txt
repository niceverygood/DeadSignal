function box(ctx, x, y, w, h, fill, stroke = null, lineWidth = 1) {
  ctx.fillStyle = fill;
  ctx.fillRect(x, y, w, h);
  if (stroke) {
    ctx.strokeStyle = stroke;
    ctx.lineWidth = lineWidth;
    ctx.strokeRect(x, y, w, h);
  }
}

function ellipse(ctx, x, y, rx, ry, fill, stroke = null, lineWidth = 1) {
  ctx.beginPath();
  ctx.ellipse(x, y, Math.max(0.01, rx), Math.max(0.01, ry), 0, 0, Math.PI * 2);
  ctx.fillStyle = fill;
  ctx.fill();
  if (stroke) {
    ctx.strokeStyle = stroke;
    ctx.lineWidth = lineWidth;
    ctx.stroke();
  }
}

// One projected world capsule, including round ends, without overlapping fills.
function capsulePath(ctx, x, y, targetX, targetY, radius, z = 1) {
  const a=project(x,y,z),b=project(targetX,targetY,z),dx=targetX-x,dy=targetY-y,distance=Math.hypot(dx,dy);
  const rx=radius*Math.sqrt(1.5),ry=radius/Math.sqrt(2);
  ctx.beginPath();
  if(distance<1e-9){ctx.ellipse(a.x,a.y,rx,ry,0,0,Math.PI*2);return;}
  const normal=project(-dy/distance*radius,dx/distance*radius),angle=Math.atan2(dy,dx)+Math.PI/4;
  ctx.moveTo(a.x+normal.x,a.y+normal.y);ctx.lineTo(b.x+normal.x,b.y+normal.y);
  ctx.ellipse(b.x,b.y,rx,ry,0,angle+Math.PI/2,angle-Math.PI/2,true);
  ctx.lineTo(a.x-normal.x,a.y-normal.y);
  ctx.ellipse(a.x,a.y,rx,ry,0,angle-Math.PI/2,angle-Math.PI*1.5,true);
  ctx.closePath();
}

function path(ctx, points, fill = null, stroke = null, lineWidth = 1, close = true) {
  ctx.beginPath();
  ctx.moveTo(points[0][0], points[0][1]);
  for (let i = 1; i < points.length; i++) ctx.lineTo(points[i][0], points[i][1]);
  if (close) ctx.closePath();
  if (fill) { ctx.fillStyle = fill; ctx.fill(); }
  if (stroke) { ctx.strokeStyle = stroke; ctx.lineWidth = lineWidth; ctx.stroke(); }
}

function line(ctx, x1, y1, x2, y2, color, width = 1) {
  ctx.beginPath(); ctx.moveTo(x1, y1); ctx.lineTo(x2, y2);
  ctx.strokeStyle = color; ctx.lineWidth = width; ctx.stroke();
}

function glow(ctx, x, y, r, color, alpha = 0.25) {
  ctx.save();
  ctx.globalAlpha *= alpha;
  const g = ctx.createRadialGradient(x, y, 0, x, y, Math.max(1, r));
  g.addColorStop(0, color); g.addColorStop(1, 'transparent');
  ctx.fillStyle = g; ctx.fillRect(x - r, y - r, r * 2, r * 2);
  ctx.restore();
}

  prepareNativeEnemyBatch(game,time,now){
    const alive=this.liveEnemySelection,corpses=this.batchNativeCorpses===false?null:this.liveCorpseSelection;
    const count=(alive?.size||0)+(corpses?.size||0);
    if(!count||!this.liveEnemies?.beginBatch||game.state!=='playing'){this.liveEnemies?.batchInk?.release();return;}
    try{
      this.liveEnemies.beginBatch(count);
      for(const e of alive||[])this.nativeEnemyFrame(e,time,game);
      // Cache each fresh death in the same atlas as the living actors. Its
      // actual draw remains in the original depth-sorted corpse pass.
      for(const corpse of corpses||[])this.nativeCorpseFrame(corpse,game,now);
      this.liveEnemies.endBatch();
    }catch(error){this.liveEnemies.endBatch();this.productionDataset({productionEnemyRigError:error.message});}
    finally{this.enemyLiveFrames=0;this.enemyLiveCorpseFrames=0;}
  }
  nativeCorpseFrame(corpse,game,now=0){
    if(!corpse)return null;
    const boss=getBoss(corpse.bossKey),section=boss?'bosses':'enemies',key=boss?corpse.bossKey:corpse.type;
    if(!key||!this.productionArt.manifest[section]?.[key])return null;
    this.requestProductionCatalog(section,key);
    let elapsed=Math.max(0,num(corpse.duration,12)-num(corpse.time,12));
    // Final victory freezes simulation immediately. Only the already defeated
    // native body completes its visual clip; results, RNG and saves stay frozen.
    if(['victory','dead'].includes(game?.state)){
      this.nativeCorpseEndClocks??=new WeakMap();
      if(!this.nativeCorpseEndClocks.has(corpse))this.nativeCorpseEndClocks.set(corpse,num(now));
      elapsed+=Math.max(0,num(now)-this.nativeCorpseEndClocks.get(corpse));
    }
    if(this.motionEnabled===false)elapsed=30;
    const options={epoch:game?.world,simulationTime:num(game?.time),pose:'death',animationTime:elapsed};
    const selected=boss?this.productionArt.bossFrame(key,corpse,options):this.productionArt.enemyFrame(key,corpse,options);
    if(!selected?.metadata.animation?.clips?.death)return null;
    const asset=this.productionArt.manifest.assets[selected.id];
    const height=num(asset?.worldHeight,selected.metadata.projection.subjectHeight);
    if(!boss){const completed=this.completedEnemyFrames?.get(corpse,height);if(completed){const foot=project(corpse.x,corpse.y);return {...completed,placement:{...spritePlacement(completed.metadata,completed.frame,{foot,worldHeight:height}),groundFoot:foot,groundingHeight:0}};}}
    if(boss&&this.liveBoss){try{const live=this.liveBoss.frame(key,corpse,num(game?.time),selected,{...options,height});if(live)return live;}catch(error){this.productionDataset({productionBossRigError:error.message});}}
    if(!boss&&this.liveEnemies&&elapsed<1.2&&this.motionEnabled!==false&&(this.enemyLiveFrames||0)<12&&(this.liveCorpseSelection?this.liveCorpseSelection.has(corpse):true))try{this.enemyLiveCorpseFrames=(this.enemyLiveCorpseFrames||0)+1;this.enemyLiveFrames=(this.enemyLiveFrames||0)+1;const live=this.liveEnemies.frame('enemy-'+key,corpse,num(game?.time),selected,{...options,height});if(live)return this.completedEnemyFrames?.remember(corpse,live,height)||live;}catch(error){this.productionDataset({productionEnemyRigError:error.message});}
    return {...selected,placement:spritePlacement(selected.metadata,selected.frame,{foot:project(corpse.x,corpse.y),worldHeight:height})};
  }

      x=towardRight?marker.x-width-20:marker.x+20;y=marker.y-height/2;align=towardRight?'right':'left';
    }
    const minX=safe.width>width+4?safe.x+2:8,maxX=safe.width>width+4?safe.x+safe.width-width-2:this.width-width-8;
    const minY=safe.height>height+4?safe.y+2:8,maxY=safe.height>height+4?safe.y+safe.height-height-2:this.height-height-8;
    return{x:clamp(x,minX,maxX),y:clamp(y,minY,maxY),width,height,align};
  }
  objectiveMarkerBounds(marker){
    const dx=Math.cos(marker.angle),dy=Math.sin(marker.angle),plaque=marker.plaque;
    const roleX=marker.x-dx*12,roleY=marker.y-dy*12,arrowX=marker.x-dx*4,arrowY=marker.y-dy*4;
    const left=Math.min(marker.x-1,roleX-6,arrowX-5.5,plaque.x-1),right=Math.max(marker.x+1,roleX+6,arrowX+5.5,plaque.x+plaque.width+1);
    const top=Math.min(marker.y-1,roleY-6,arrowY-5.5,plaque.y-1),bottom=Math.max(marker.y+1,roleY+6,arrowY+5.5,plaque.y+plaque.height+1);
    return{left,top,right,bottom,x:left,y:top,width:right-left,height:bottom-top};
  }
  objectivePlacementClear(marker,placed){
    const box=marker.bounds;
    if(box.left<5||box.top<5||box.right>this.width-5||box.bottom>this.height-5)return false;
    const overlap=r=>r&&Number.isFinite(r.left)&&Number.isFinite(r.top)&&Number.isFinite(r.right)&&Number.isFinite(r.bottom)&&r.right>r.left&&r.bottom>r.top&&box.right>r.left-6&&box.left<r.right+6&&box.bottom>r.top-6&&box.top<r.bottom+6;
    for(const rect of this.directionExclusions||[])if(overlap(rect))return false;
    for(const previous of placed)if(overlap(previous.bounds))return false;
    return true;
  }
  objectiveDirectionMarkers(game){
    if(game.state!=='playing'||!game.player)return[];
    const safe=this.objectiveSafeRect(game),player=this.worldToScreen(game.player.x,game.player.y,22),markers=[];
    for(const target of objectiveTargets(game)){
      const point=this.worldToScreen(target.enemy.x,target.enemy.y,22),position=edgePosition(player,point,this.width,this.height,safe);
      if(!position)continue;
      const vertical=Math.abs(position.x-safe.x)<1||Math.abs(position.x-safe.x-safe.width)<1;
      const build=(x,y)=>{const marker={kind:target.kind,label:target.label,x,y,angle:position.angle};marker.plaque=this.objectivePlaque(marker,safe);marker.bounds=this.objectiveMarkerBounds(marker);return marker;};
      let chosen=null;
      // Try the requested edge first, moving the optional mark by 34px steps.
      for(const offset of [0,34,-34,68,-68,102,-102,136,-136]){
        const x=vertical?position.x:clamp(position.x+offset,safe.x,safe.x+safe.width),y=vertical?clamp(position.y+offset,safe.y,safe.y+safe.height):position.y;
        const candidate=build(x,y);if(this.objectivePlacementClear(candidate,markers)){chosen=candidate;break;}
      }
      // A crowded compact HUD can leave no free edge. Keep the original direction
      // while looking for a small empty middle position, rather than cover controls.
      if(!chosen)for(const fy of [.25,.75,.5]){
        for(const fx of [.5,.25,.75]){
          const candidate=build(safe.x+safe.width*fx,safe.y+safe.height*fy);
          if(this.objectivePlacementClear(candidate,markers)){chosen=candidate;break;}
        }
        if(chosen)break;
      }
      if(chosen)markers.push(chosen);
    }
    return markers;
  }
  drawObjectiveDirections(c,game){
    for(const marker of this.objectiveDirectionMarkers(game)){
      const color=['gatekeeper','expedition'].includes(marker.kind)?'#e9c16f':marker.kind==='collector'?'#dc7866':'#e18a71',plaque=marker.plaque;
      c.save();c.translate(marker.x,marker.y);c.rotate(marker.angle);
      path(c,[[0,0],[-6,-3.5],[-4,0],[-6,3.5]],color,'#18313b',1.1);c.restore();
      const x=marker.x-Math.cos(marker.angle)*12,y=marker.y-Math.sin(marker.angle)*12;
      c.save();
      // Role symbols stay upright, matching the map; only the arrow changes heading.
      if(marker.kind==='collector'){
        const hex=[];for(let i=0;i<6;i++){const a=i*Math.PI/3+Math.PI/6;hex.push([x+Math.cos(a)*5,y+Math.sin(a)*5]);}
        path(c,hex,'#b9554a','#f0e1bb',1);
      }else if(marker.kind==='gatekeeper'){
        path(c,[[x,y-4],[x+4,y],[x,y+4],[x-4,y]],color,'#293c33',.8);
      }else{
        path(c,[[x,y-4.5],[x+4,y+3],[x-4,y+3]],color,'#50352f',.8);
      }
      box(c,plaque.x,plaque.y,plaque.width,plaque.height,'rgba(14,29,36,.92)',color,.65);
      c.font='700 10px "Noto Sans KR",sans-serif';c.fillStyle=color;c.textAlign=plaque.align;c.textBaseline='middle';
      const textX=plaque.align==='left'?plaque.x+6:plaque.align==='right'?plaque.x+plaque.width-6:plaque.x+plaque.width/2;
      c.fillText(marker.label,textX,plaque.y+plaque.height/2+.5,plaque.width-12);c.restore();
    }
  }
  drawExit(c,exit,time,stage,presentation=null){
    if(!this.visible(exit.x,exit.y,170))return;
    const x=exit.x,y=exit.y,active=presentation?.open??Boolean(exit.open),color=active?'#b4ddc3':'#b4a16e';
    this.polygon(c,[[x-70,y-55],[x+70,y-55],[x+70,y+45],[x-70,y+45]],0,'#3b535b','#7e948b',2);
    for(let i=0;i<6;i++)this.polygon(c,[[x-67+i*24,y+43],[x-56+i*24,y+43],[x-44+i*24,y+31],[x-55+i*24,y+31]],1,'#c8b574');
    this.polygon(c,[[x-4,y-38],[x+19,y-12],[x+7,y-12],[x+7,y+13],[x-7,y+13],[x-7,y-12],[x-19,y-12]],1,active?'#a9cfb5':'#748277');
    for(const xx of [x-70,x+70]){this.cube(c,xx-5,y-60,10,10,45,'#607b72','#3b5a60','#26444f');const p=project(xx,y-56,48);ellipse(c,p.x,p.y,4,2,active?'#a4dfb9':'#ddae5d');if(active)glow(c,p.x,p.y,25,'#96dab4',.25);}
    const p=project(x,y-52,51);c.font='700 9px monospace';c.fillStyle=color;c.textAlign='center';c.fillText(presentation?.label||(active?(stage===2?'EXTRACTION / OPEN':'NEXT SECTOR / OPEN'):'GATE / LOCKED'),p.x,p.y);
  }
  prepareMinimap(game){
    const world=game.world,key=`${layoutKey(world)}:${num(game.stage)}:${this.dpr}`;
    if(this.minimapCanvas&&this.minimapKey===key)return;
    this.minimapKey=key;
    const width=178,height=160,canvas=document.createElement('canvas');
    canvas.width=Math.round(width*this.dpr);canvas.height=Math.round(height*this.dpr);
    const c=canvas.getContext('2d');c.setTransform(this.dpr,0,0,this.dpr,0,0);c.lineJoin='round';
    const m=makeMinimapProjection(world,{x:10,y:27,width:158,height:113,padding:4});
    this.minimapCanvas=canvas;this.minimapProjection=m;
    box(c,0,0,width,height,'rgba(13,27,35,.94)','#5f777a',1);
    line(c,0,0,35,0,'#d9bd78',2);
    c.font='700 8px monospace';c.textAlign='left';c.fillStyle='#d0d1b8';
    c.fillText(`${String(num(game.missionIndex,num(game.stage))+1).padStart(2,'0')} / SIGNAL MAP`,10,16);
    c.font='700 7px monospace';c.fillStyle='#83988f';c.fillText('M',161,16);
    c.font='700 8px "Noto Sans KR",sans-serif';c.fillStyle='#bac5b4';
    c.fillText(game.missionId?'중간':'봉쇄',20,154);c.fillText('경보',73,154);c.fillText(game.missionId?'최종':'수집가',127,154);
    path(c,[[12,147],[16,151],[12,155],[8,151]],'#e9c16f','#3e4636',.6);
    path(c,[[64,147],[68,154],[60,154]],'#e18a71','#57372e',.6);
    const legendHex=[];for(let i=0;i<6;i++){const a=i*Math.PI/3+Math.PI/6;legendHex.push([117+Math.cos(a)*3.8,151+Math.sin(a)*3.8]);}
    path(c,legendHex,'#b9554a','#f0e1bb',.8);
    path(c,m.corners.map(p=>[p.x,p.y]),'#243b43','#718174',.7);
    const footprint=(x,y,w,h,fill,stroke='#274348',weight=.5)=>{
      const corners=[m.project(x,y),m.project(x+w,y),m.project(x+w,y+h),m.project(x,y+h)];
      path(c,corners.map(p=>[p.x,p.y]),fill,stroke,weight);
    };
    for(const road of world.roads||[])footprint(road.x,road.y,road.w,road.h,'#38535c',null);
    for(const water of world.water||[])footprint(water.x,water.y,water.w,water.h,'#102e3d','#57808d',.5);
    for(const o of world.obstacles||[]){
      if(o.destroyed)continue;
      const color={building:'#768372',car:'#6c8787',barrel:'#a28d5b',crate:'#83956d',fence:'#647c74'}[o.kind]||'#718675';
      footprint(o.x,o.y,o.w,o.h,color,o.kind==='building'?'#233d43':'#3a575a',o.kind==='building'?.6:.4);
    }
  }
  drawMinimap(c,game){
    if(!this.minimapEnabled||this.width<900||this.height<560||game.state!=='playing'||!game.world||!game.player)return;
    this.prepareMinimap(game);
    const width=178,height=160,x=this.width-width-26,y=140,m=this.minimapProjection;
    c.save();c.translate(x,y);c.drawImage(this.minimapCanvas,0,0,width,height);c.lineJoin='round';
    // A faint camera footprint helps players relate the fixed city plan to their view.
    c.save();path(c,m.corners.map(p=>[p.x,p.y]));c.clip();
    const viewport=[];
    for(const [sx,sy]of [[0,0],[this.width,0],[this.width,this.height],[0,this.height]]){
      const w=this.screenToWorld(sx,sy),p=m.project(w.x,w.y);viewport.push([p.x,p.y]);
    }
    path(c,viewport,null,'rgba(213,220,184,.25)',.65);c.restore();
    const alarmId=game.alarm?.eliteId;
    const alive=e=>!e.dead&&num(e.hp,1)>0&&Number.isFinite(e.x)&&Number.isFinite(e.y);
    const special=e=>e.type==='collector'||e.gatekeeper||(alarmId!==null&&alarmId!==undefined&&e.id===alarmId);
    // Ordinary horde dots go first so mandatory and optional objective markers stay clear.
    for(const e of game.enemies||[])if(alive(e)&&!special(e)){
      const p=m.project(e.x,e.y);ellipse(c,p.x,p.y,1.1,1.1,'#ca705c');
    }
    for(const e of game.enemies||[])if(alive(e)&&special(e)){
      const p=m.project(e.x,e.y);
      if(e.type==='collector'){
        const hex=[];for(let i=0;i<6;i++){const a=i*Math.PI/3+Math.PI/6;hex.push([p.x+Math.cos(a)*5.2,p.y+Math.sin(a)*5.2]);}
        path(c,hex,'#b9554a','#f0e1bb',1.05);
      }else if(e.gatekeeper){
        path(c,[[p.x,p.y-3.8],[p.x+3.8,p.y],[p.x,p.y+3.8],[p.x-3.8,p.y]],'#e9c16f','#3e4636',.8);
      }else{
        path(c,[[p.x,p.y-4.2],[p.x+3.8,p.y+3.2],[p.x-3.8,p.y+3.2]],'#e18a71','#57372e',.8);
      }
    }
    if(game.lure){
      const p=m.project(game.lure.x,game.lure.y);
      ellipse(c,p.x,p.y,3.8,3.8,'rgba(232,179,73,.16)','#edbf68',1);ellipse(c,p.x,p.y,1.2,1.2,'#edc57b');
    }
    if(game.world.exit){
      const e=game.world.exit,p=m.project(e.x,e.y),active=exitPresentation(game).open,color=active?'#a3d9b4':'#8e9885';
      box(c,p.x-3.2,p.y-3.2,6.4,6.4,'#203d43',color,1.1);
      line(c,p.x,p.y-1.8,p.x,p.y+1.8,color,1.2);
      if(active)line(c,p.x-1.8,p.y,p.x+1.8,p.y,color,1.1);
    }
    const player=game.player,p=m.project(player.x,player.y);
    c.translate(p.x,p.y);c.rotate(m.heading(num(player.angle)));
    path(c,[[5.3,0],[-3.9,-3.1],[-2.1,0],[-3.9,3.1]],'#f3e5b9','#102b35',1.2);
    c.restore();
  }
